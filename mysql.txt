CREATE DATABASE IF NOT EXISTS campus_connect;
USE campus_connect;


-- ============================================================
-- 1. DEPARTMENTS TABLE
-- ============================================================

CREATE TABLE IF NOT EXISTS departments (
    dept_id INT AUTO_INCREMENT PRIMARY KEY,
    dept_name VARCHAR(150) NOT NULL,
    dept_code VARCHAR(20),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);


-- ============================================================
-- 2. SECTIONS TABLE
-- ============================================================

CREATE TABLE IF NOT EXISTS sections (
    section_id INT AUTO_INCREMENT PRIMARY KEY,
    dept_id INT NOT NULL,
    section_name CHAR(1) NOT NULL,
    section_order INT,
    academic_year INT DEFAULT 2,
    student_capacity INT DEFAULT 60
);


-- ============================================================
-- 3. SUBJECTS TABLE
-- ============================================================

CREATE TABLE IF NOT EXISTS subjects (
    subject_id INT AUTO_INCREMENT PRIMARY KEY,
    dept_id INT NOT NULL,
    subject_code VARCHAR(30) NOT NULL UNIQUE,
    subject_name VARCHAR(150) NOT NULL,
    subject_order INT
);


-- ============================================================
-- 4. STUDENTS TABLE
-- ============================================================

CREATE TABLE IF NOT EXISTS students (
    student_id INT AUTO_INCREMENT PRIMARY KEY,
    student_code VARCHAR(30),
    roll_number VARCHAR(30),
    student_name VARCHAR(100) NOT NULL,
    dept_id INT NOT NULL,
    section_id INT,
    year_of_study INT DEFAULT 2,
    semester INT DEFAULT 3,
    mobile_number VARCHAR(20)
);


-- ============================================================
-- 5. FACULTY TABLE
-- ============================================================

CREATE TABLE IF NOT EXISTS faculty (
    faculty_id INT AUTO_INCREMENT PRIMARY KEY,
    faculty_name VARCHAR(100) NOT NULL,
    dept_id INT NOT NULL,
    designation VARCHAR(100),
    email VARCHAR(100),
    mobile_number VARCHAR(20)
);


-- ============================================================
-- 6. USER ACCOUNTS TABLE
-- ============================================================

CREATE TABLE IF NOT EXISTS user_accounts (
    user_id INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(100) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    role ENUM(
        'STUDENT',
        'FACULTY',
        'HOD',
        'OPERATOR',
        'ADMIN'
    ) NOT NULL,
    student_id INT UNIQUE NULL,
    faculty_id INT UNIQUE NULL,
    is_active TINYINT(1) DEFAULT 1
);


-- ============================================================
-- 7. ATTENDANCE TABLE
-- ============================================================

CREATE TABLE IF NOT EXISTS attendance (
    attendance_id INT AUTO_INCREMENT PRIMARY KEY,
    student_id INT NOT NULL,
    subject_id INT NOT NULL,
    total_classes INT NOT NULL DEFAULT 0,
    attended_classes INT NOT NULL DEFAULT 0,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP
);


-- ============================================================
-- 8. FACULTY ANNOUNCEMENTS TABLE
-- ============================================================

CREATE TABLE IF NOT EXISTS faculty_announcements (
    announcement_id INT AUTO_INCREMENT PRIMARY KEY,
    faculty_id INT NOT NULL,
    dept_id INT NOT NULL,
    year_of_study INT NULL,
    subject_id INT NULL,
    title VARCHAR(200) NOT NULL,
    message TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);


-- ============================================================
-- 9. CLUBS TABLE
-- ============================================================

CREATE TABLE IF NOT EXISTS clubs (
    club_id INT AUTO_INCREMENT PRIMARY KEY,
    club_name VARCHAR(100) NOT NULL UNIQUE,
    description TEXT,
    coordinator VARCHAR(100),
    is_active TINYINT(1) DEFAULT 1
);


-- ============================================================
-- 10. EVENTS TABLE
-- ============================================================

CREATE TABLE IF NOT EXISTS events (
    event_id INT AUTO_INCREMENT PRIMARY KEY,
    event_name VARCHAR(150) NOT NULL,
    description TEXT,
    event_date DATE,
    venue VARCHAR(150),
    created_by INT,
    is_active TINYINT(1) DEFAULT 1
);


-- ============================================================
-- 11. CLUB MEMBERSHIPS TABLE
-- ============================================================

CREATE TABLE IF NOT EXISTS club_memberships (
    membership_id INT AUTO_INCREMENT PRIMARY KEY,
    student_id INT NOT NULL,
    club_id INT NOT NULL,
    membership_type VARCHAR(30)
);


-- ============================================================
-- 12. EVENT REGISTRATIONS TABLE
-- ============================================================

CREATE TABLE IF NOT EXISTS event_registrations (
    registration_id INT AUTO_INCREMENT PRIMARY KEY,
    event_id INT NOT NULL,
    student_id INT NOT NULL,
    registered_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);


-- ============================================================
-- 13. FEEDBACK TABLE
-- ============================================================

CREATE TABLE IF NOT EXISTS feedback (
    feedback_id INT AUTO_INCREMENT PRIMARY KEY,
    student_id INT NOT NULL,
    feedback_message TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 31. VIEW DEPARTMENTS

SELECT * FROM departments
ORDER BY dept_id;


-- ============================================================
-- 32. VIEW SECTIONS
-- ============================================================

SELECT * FROM sections
ORDER BY dept_id, section_order;


-- ============================================================
-- 33. SUBJECT COUNT BY DEPARTMENT
-- ============================================================

SELECT
    d.dept_id,
    d.dept_name,
    COUNT(s.subject_id) AS total_subjects
FROM departments d
LEFT JOIN subjects s
    ON s.dept_id = d.dept_id
GROUP BY d.dept_id, d.dept_name
ORDER BY d.dept_id;


-- ============================================================
-- 34. VIEW STUDENTS
-- ============================================================

SELECT * FROM students;


-- ============================================================
-- 35. VIEW FACULTY

SELECT * FROM faculty;


-- ============================================================
-- 36. VIEW EXISTING LOGIN ACCOUNTS
-- Never display password hashes unnecessarily.
-- ============================================================

SELECT
    username,
    role,
    student_id,
    faculty_id,
    is_active
FROM user_accounts
ORDER BY role, username;


-- ============================================================
-- 37. SUBJECT-WISE ATTENDANCE
-- ============================================================

SELECT
    s.student_name,
    sub.subject_name,
    a.total_classes,
    a.attended_classes,
    ROUND(
        a.attended_classes * 100.0 /
        NULLIF(a.total_classes, 0),
        2
    ) AS attendance_percentage
FROM attendance a
JOIN students s
    ON s.student_id = a.student_id
JOIN subjects sub
    ON sub.subject_id = a.subject_id;


-- ============================================================
-- 38. VIEW FACULTY ANNOUNCEMENTS
-- ============================================================

SELECT
    a.announcement_id,
    a.title,
    a.message,
    a.dept_id,
    a.year_of_study,
    a.subject_id,
    f.faculty_name
FROM faculty_announcements a
JOIN faculty f
    ON f.faculty_id = a.faculty_id
ORDER BY a.created_at DESC;


-- ============================================================
-- 39. STUDENT FACULTY MESSAGES
-- Example: AI department (6), year 2
-- ============================================================

SELECT
    a.announcement_id,
    a.title,
    a.message,
    a.created_at,
    f.faculty_name,
    f.designation,
    f.email,
    f.mobile_number,
    sub.subject_name,
    sub.subject_code
FROM faculty_announcements a
JOIN faculty f
    ON f.faculty_id = a.faculty_id
LEFT JOIN subjects sub
    ON sub.subject_id = a.subject_id
WHERE a.dept_id = 6
AND (
    a.year_of_study IS NULL
    OR a.year_of_study = 2
)
ORDER BY a.created_at DESC;

-- 42. VIEW STUDENT FEEDBACK

SELECT
    f.feedback_id,
    s.student_name,
    f.feedback_message,
    f.created_at
FROM feedback f
JOIN students s
    ON s.student_id = f.student_id
ORDER BY f.created_at DESC;
